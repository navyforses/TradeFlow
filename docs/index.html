strategy.py
from tradeflow.api import order_target, record, symbol
from tradeflow.indicators import sma, rsi

def initialize(context):
    context.asset = symbol('AAPL')
    context.lookback = 20

def handle_data(context, data):
    prices = data.history(context.asset, 'price', 50, '1d')
    signal = sma(prices, 20).iloc[-1]
    if prices.iloc[-1] > signal:
        order_target(context.asset, 100)
    record(price=prices.iloc[-1], sma=signal)
